@'
import hashlib,json
from pathlib import Path
from PIL import Image
root=Path(r"D:\cat-mouse-miniprogram\design\2026-10-11\page-concepts-v1")
m=json.loads((root/"screens.json").read_text(encoding="utf-8"))
assert len(m["screens"])==16
for c in m["screens"]:
    p=root/c["file"]
    with Image.open(p) as im:
        assert im.format=="PNG"
        im.verify()
    with Image.open(p) as im:
        im.load()
        c["width"],c["height"]=im.size
        c["mode"]=im.mode
    assert (c["width"],c["height"])==(1024,1536),(p,im.size)
    c["bytes"]=p.stat().st_size
    c["sha256"]=hashlib.sha256(p.read_bytes()).hexdigest()
    assert c["sha256"]==hashlib.sha256(Path(c["source"]).read_bytes()).hexdigest(),p
    for call in c["calls"]:
        assert Path(call["source"]).is_file()
        for ref in call["referenced_image_paths"]:
            assert Path(ref).is_file(),ref
    for old in c["superseded"]:
        q=Path(old["absolute_path"])
        with Image.open(q) as im: im.verify()
        old["sha256"]=hashlib.sha256(q.read_bytes()).hexdigest()
        assert old["sha256"]==hashlib.sha256(Path(old["source"]).read_bytes()).hexdigest()
(root/"screens.json").write_text(json.dumps(m,ensure_ascii=False,indent=2)+"\n",encoding="utf-8")
print("PASS: 16 selected PNGs decode; all 1024x1536; SHA-256 matches original outputs")
print("PASS: 2 superseded PNGs preserved and hash-matched; all input references exist")
print("PASS: 16 generation prompts and 2 edit prompts archived")
